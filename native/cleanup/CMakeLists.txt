cmake_minimum_required(VERSION 3.22)
project(velora_cleanup_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/UPSTREAM.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/UPSTREAM.json" UPSTREAM_JSON)
string(JSON LLAMA_PIN GET "${UPSTREAM_JSON}" commit)
configure_file(build_pin.h.in "${CMAKE_CURRENT_BINARY_DIR}/build_pin.h" @ONLY)
set(LLAMA_SOURCE "" CACHE PATH "Verified llama.cpp checkout; see UPSTREAM.json")
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
  message(FATAL_ERROR "Set LLAMA_SOURCE to the pinned local source")
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py"
  "${LLAMA_SOURCE}" "${CMAKE_CURRENT_SOURCE_DIR}/UPSTREAM.json"
  RESULT_VARIABLE PIN_RESULT OUTPUT_VARIABLE PIN_OUTPUT ERROR_VARIABLE PIN_ERROR)
if(NOT PIN_RESULT EQUAL 0)
  message(FATAL_ERROR "${PIN_ERROR}")
endif()
message(STATUS "${PIN_OUTPUT}")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
add_custom_target(verify-llama-source ALL
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py"
    "${LLAMA_SOURCE}" "${CMAKE_CURRENT_SOURCE_DIR}/UPSTREAM.json" VERBATIM)
add_subdirectory("${LLAMA_SOURCE}" llama-build)
add_dependencies(llama verify-llama-source)
add_dependencies(llama-common verify-llama-source)
add_executable(velora-cleanup-native worker.cpp "${LLAMA_SOURCE}/vendor/hash/sha256/sha256.c")
add_dependencies(velora-cleanup-native verify-llama-source)
target_include_directories(velora-cleanup-native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${LLAMA_SOURCE}/vendor" "${LLAMA_SOURCE}/vendor/hash/sha256" "${LLAMA_SOURCE}/vendor/hash")
target_link_libraries(velora-cleanup-native PRIVATE llama llama-common)
